# =============================================================================
# FUNCIÓN DE VALIDACIÓN
# =============================================================================
def validar_dataset(datos, L0, A, nombre_categoria):
    """Valida un dataset contra el modelo teórico."""
    resultados = []
    
    for nombre, masa in datos.items():
        razon_log = math.log(masa / L0)
        q = round(razon_log / math.log(A))
        masa_pred = L0 * (A ** q)
        error = abs(masa - masa_pred) / masa * 100
        
        resultados.append({
            "Molecula": nombre,
            "Masa_Real": masa,
            "q": q,
            "Masa_Pred": masa_pred,
            "Error_%": error,
            "Categoria": nombre_categoria
        })
    
    return pd.DataFrame(resultados)

def estadisticas_categoria(df, nombre):
    """Calcula estadísticas para una categoría."""

        "N": len(df),
        "Error Medio (%)": df["Error_%"].mean(),
        "Error Mediano (%)": df["Error_%"].median(),
        "Std Dev (%)": df["Error_%"].std(),
        "Máx Error (%)": df["Error_%"].max(),
        "Mín Error (%)": df["Error_%"].min(),
        "≤5%": (df["Error_%"] <= 5).sum(),
        "≤10%": (df["Error_%"] <= 10).sum(),
        ">10%": (df["Error_%"] > 10).sum(),
    }
